    95→    95→    95→ * كل محاولة (نجاحًا أو فشلًا) تُسجَّل في attempts_json وتحدّث عدّادات المفتاح.

    98→    98→    98→async function tryCreateWithKeys(opts: {
    99→    99→    99→  task: TaskRow;
   100→   100→   100→  modelDef: ModelDef;
   101→   101→   101→  input: ImageInput | VideoInput;
   102→   102→   102→  skipIds?: Set<string>;
   103→   103→   103→}): Promise<CreateAttemptOutcome> {
   104→   104→   104→  const { task, modelDef, input, skipIds } = opts;
   105→   105→   105→  const pool = await getKeyPool(modelDef.providers);
   106→   106→   106→  const candidates = pool.filter((k) => !skipIds?.has(k.id) && modelDef.providers.includes(k.provider as ProviderId));
   107→   107→   107→
   108→   108→   108→  for (const key of candidates) {
   109→   109→   109→    const adapter = getAdapter(key.provider as ProviderId);
   110→   110→   110→    const providerModel = modelDef.providerModel[key.provider] ?? modelDef.canonical;
   111→   111→   111→    try {
   112→   112→   112→      // صور متزامنة إن دعم المزود ذلك (CometAPI)
   113→   113→   113→      if (task.type === "image" && adapter.createImageSync) {
   114→   114→   114→        const { urls } = await adapter.createImageSync(key, input as ImageInput);
   115→   115→   115→        await bumpKeyCounters(key.id, true);
   116→   116→   116→        // مفتاح كان مستنزفًا وعاد للعمل (شُحن رصيده) → نعيده active
   117→   117→   117→        if (key.status === "exhausted") await setKeyStatus(key.id, "active", null);
   118→   118→   118→        await recordAttempt(task.id, { provider: key.provider, keyId: key.id, ok: true });
   119→   119→   119→        return { kind: "created_sync", key, syncUrls: urls };
   120→   120→   120→      }
   121→   121→   121→      const { providerTaskId } = await adapter.createTask(key, providerModel, input);
   122→   122→   122→      await bumpKeyCounters(key.id, true);
   123→   123→   123→      if (key.status === "exhausted") await setKeyStatus(key.id, "active", null);
   124→   124→   124→      await recordAttempt(task.id, { provider: key.provider, keyId: key.id, ok: true });
   125→   125→   125→      return { kind: "created", key, providerTaskId };
   126→   126→   126→    } catch (err) {
   127→   127→   127→      const cls = classifyProviderError(err);
   128→   128→   128→      const msg = err instanceof Error ? err.message : String(err);
   129→   129→   129→      await bumpKeyCounters(key.id, false, msg.slice(0, 240));
   130→   130→   130→      await recordAttempt(task.id, {
   131→   131→   131→        provider: key.provider, keyId: key.id, ok: false,
   132→   132→   132→        errorClass: cls, error: msg.slice(0, 240),
   133→   133→   133→      });
   134→   134→   134→      if (cls === "params") {
   135→   135→   135→        // خطأ معطيات — توقف فورًا: تبديل المفاتيح لن يحلّه
   136→   136→   136→        return { kind: "params", paramsMessage: sanitize(msg) };
   137→   137→   137→      }
   138→   138→   138→      if (cls === "disabled") {
   139→   139→   139→        await setKeyStatus(key.id, "disabled", msg.slice(0, 240)); // مفتاح مرفوض → تعطيل تلقائي
   140→   140→   140→        continue; // → المفتاح التالي
   141→   141→   141→      }
   142→   142→   142→      if (cls === "exhausted") {
   143→   143→   143→        await setKeyStatus(key.id, "exhausted", msg.slice(0, 240)); // يبقى في القاعدة
   144→   144→   144→        continue; // → المفتاح التالي
   145→   145→   145→      }
   146→   146→   146→      // provider (موديل غير مدعوم) أو temporary → التالي
   147→   147→   147→      continue;
   148→   148→   148→    }
   149→   149→   149→  }
   150→   150→   150→  return { kind: "exhausted" };
   151→   151→   151→}

   161→   161→   161→export async function runGeneration(
   162→   162→   162→  type: "image" | "video",
   163→   163→   163→  args: Record<string, unknown>,
   164→   164→   164→  notify?: NotifyFn
   165→   165→   165→): Promise<ToolOutcome> {
   166→   166→   166→  const modelDef = resolveModel(args.model as string | undefined);
   167→   167→   167→  if (!modelDef || modelDef.type !== type) {
   168→   168→   168→    return {
   169→   169→   169→      status: "params",
   170→   170→   170→      isError: true,
   171→   171→   171→      text: fmtFailed(`موديل غير معروف: ${String(args.model ?? "")}. المتاح: gpt-image-2-6/flare-text-to-image (صور) و grok-imagine-video-1-5-preview (فيديو).`),
   172→   172→   172→    };
   173→   173→   173→  }
   174→   174→   174→
   175→   175→   175→  const pool = await getKeyPool(modelDef.providers);
   176→   176→   176→  if (!pool.length) return { status: "no_keys", isError: true, text: fmtNoKeys(modelDef.canonical) };
   177→   177→   177→
   178→   178→   178→  // 3) سجّل المهمة قبل أي نداء خارجي
   179→   179→   179→  const input = type === "image" ? buildImageInput(args) : buildVideoInput(args);
   180→   180→   180→  let task = await insertTask({
   181→   181→   181→    id: randomUUID(),
   182→   182→   182→    model: modelDef.canonical,
   183→   183→   183→    type,
   184→   184→   184→    prompt: input.prompt,
   185→   185→   185→    params_json: JSON.stringify(args),
   186→   186→   186→    status: "processing",
   187→   187→   187→  });
   188→   188→   188→
   189→   189→   189→  const waitSec = effectiveWait(args.wait_seconds as number | undefined, type === "image" ? 90 : 25);
   190→   190→   190→  let deadline = Date.now() + waitSec * 1000;
   191→   191→   191→
   192→   192→   192→  // 4) تجربة المرشحين بالترتيب حتى نجاح الإنشاء
   193→   193→   193→  let created = await tryCreateWithKeys({ task, modelDef, input });
   194→   194→   194→  if (created.kind === "params") {
   195→   195→   195→    task = (await updateTask(task.id, { status: "failed", error_message: created.paramsMessage }))!;
   196→   196→   196→    return { status: "failed", isError: true, text: fmtFailed(created.paramsMessage!), task };
   197→   197→   197→  }
   198→   198→   198→  if (created.kind === "exhausted") {
   199→   199→   199→    // تبقى processing — تُعاد المحاولة مع كل get_task بعد التهدئة (لا شيء يُذكر للعميل)
   200→   200→   200→    task = (await updateTask(task.id, { error_message: "بانتظار توفر مفتاح صالح…" }))!;
   201→   201→   201→    return { status: "processing", text: fmtProcessing(task.id), task };
   202→   202→   202→  }
   203→   203→   203→
   204→   204→   204→  if (created.kind === "created_sync") {
   205→   205→   205→    task = (await updateTask(task.id, {
   206→   206→   206→      provider: created.key!.provider, key_id: created.key!.id,
   207→   207→   207→      status: "completed", result_urls_json: JSON.stringify(created.syncUrls),
   208→   208→   208→    }))!;
   209→   209→   209→    notify?.(100, 100, "اكتمل التوليد").catch(() => {});
   210→   210→   210→    return { status: "completed", text: fmtImageDone(created.syncUrls!), task };
   211→   211→   211→  }
   212→   212→   212→
   213→   213→   213→  // نجح الإنشاء — اربط المهمة بالمزود والمفتاح
   214→   214→   214→  task = (await updateTask(task.id, {
   215→   215→   215→    provider: created.key!.provider,
   216→   216→   216→    provider_task_id: created.providerTaskId,
   217→   217→   217→    key_id: created.key!.id,
   218→   218→   218→    progress: 5,
   219→   219→   219→  }))!;
   220→   220→   220→  let providerTaskId = created.providerTaskId!;
   221→   221→   221→  let adapter = getAdapter(created.key!.provider as ProviderId);
   222→   222→   222→  let currentKey = created.key!;
   223→   223→   223→
   224→   224→   224→  // 5) نافذة الانتظار: استعلام كل 3 ثوانٍ مع بث تقدم حي
   225→   225→   225→  while (Date.now() < deadline) {
   226→   226→   226→    let check;
   227→   227→   227→    try {
   228→   228→   228→      check = await adapter.checkTask(currentKey, providerTaskId);
   229→   229→   229→    } catch (err) {
   230→   230→   230→      // أخطاء شبكتنا نحن (timeout/fetch) = مؤقتة → استمر بالاستعلام
   231→   231→   231→      if (!(err instanceof ProviderError) || err.status !== 0) {
   232→   232→   232→        // خطأ مصنَّف من المزود أثناء الاستعلام: عالجه كفشل منتصف مهمة
   233→   233→   233→        check = { status: "failed", error: err instanceof Error ? err.message : String(err) };
   234→   234→   234→      } else {
   235→   235→   235→        await sleep(POLL_INTERVAL_MS);
   236→   236→   236→        continue;
   237→   237→   237→      }
   238→   238→   238→    }
   239→   239→   239→
   240→   240→   240→    if (check.status === "processing") {
   241→   241→   241→      const p = check.progress ?? task.progress;
   242→   242→   242→      if (Number.isFinite(p) && p > task.progress) task = (await updateTask(task.id, { progress: Math.round(p) })) ?? task;
   243→   243→   243→      notify?.(Math.round(check.progress ?? task.progress), 100, `قيد التوليد… ${Math.round(check.progress ?? task.progress)}%`).catch(() => {});
   244→   244→   244→      await sleep(POLL_INTERVAL_MS);
   245→   245→   245→      continue;
   246→   246→   246→    }
   247→   247→   247→
   248→   248→   248→    if (check.status === "completed" && check.urls?.length) {
   249→   249→   249→      task = (await updateTask(task.id, { status: "completed", result_urls_json: JSON.stringify(check.urls) }))!;
   250→   250→   250→      notify?.(100, 100, "اكتمل التوليد").catch(() => {});
   251→   251→   251→      return {
   252→   252→   252→        status: "completed",
   253→   253→   253→        text: type === "image" ? fmtImageDone(check.urls) : fmtVideoDone(check.urls),
   254→   254→   254→        task,
   255→   255→   255→      };
   256→   256→   256→    }
   257→   257→   257→
   258→   258→   258→    if (check.status === "failed") {
   259→   259→   259→      const msg = String(check.error ?? "");
   260→   260→   260→      if (looksLikeKeyIssue(msg)) {
   261→   261→   261→        // فشل مفتاح/رصيد في منتصف المهمة → أعد إنشاء نفس المهمة بنفس task_id على المفتاح التالي
   262→   262→   262→        const skip = new Set<string>([currentKey.id]);
   263→   263→   263→        const retry = await tryCreateWithKeys({ task, modelDef, input, skipIds: skip });
   264→   264→   264→        if (retry.kind === "created" || retry.kind === "created_sync") {
   265→   265→   265→          task = (await updateTask(task.id, {
   266→   266→   266→            provider: retry.key!.provider,
   267→   267→   267→            key_id: retry.key!.id,
   268→   268→   268→            provider_task_id: retry.providerTaskId ?? task.provider_task_id,
   269→   269→   269→            error_message: null,
   270→   270→   270→            progress: 5,
   271→   271→   271→          }))!;
   272→   272→   272→          if (retry.kind === "created_sync") {
   273→   273→   273→            task = (await updateTask(task.id, { status: "completed", result_urls_json: JSON.stringify(retry.syncUrls) }))!;
   274→   274→   274→            return { status: "completed", text: fmtImageDone(retry.syncUrls!), task };
   275→   275→   275→          }
   276→   276→   276→          providerTaskId = retry.providerTaskId!;
   277→   277→   277→          adapter = getAdapter(retry.key!.provider as ProviderId);
   278→   278→   278→          currentKey = retry.key!;
   279→   279→   279→          deadline = Date.now() + waitSec * 1000; // نافذة جديدة للمحاولة الجديدة
   280→   280→   280→          continue;
   281→   281→   281→        }
   282→   282→   282→        if (retry.kind === "params") {
   283→   283→   283→          task = (await updateTask(task.id, { status: "failed", error_message: retry.paramsMessage }))!;
   284→   284→   284→          return { status: "failed", isError: true, text: fmtFailed(retry.paramsMessage!), task };
   285→   285→   285→        }
   286→   286→   286→        // لا مفتاح متاح الآن → تبقى processing بصمت
   287→   287→   287→        task = (await updateTask(task.id, { error_message: "بانتظار توفر مفتاح صالح…" }))!;
   288→   288→   288→        return { status: "processing", text: fmtProcessing(task.id), task };
   289→   289→   289→      }
   290→   290→   290→      // فشل حقيقي (انتهاك سياسة/معطيات) → failed
   291→   291→   291→      task = (await updateTask(task.id, { status: "failed", error_message: sanitize(msg) }))!;
   292→   292→   292→      return { status: "failed", isError: true, text: fmtFailed(sanitize(msg)), task };
   293→   293→   293→    }
   294→   294→   294→  }
   295→   295→   295→
   296→   296→   296→  // 7) انتهت النافذة والنتيجة لم تظهر → أعد المهمة بحالة processing ليتابعها العميل
   297→   297→   297→  return { status: "processing", text: fmtProcessing(task.id, task.progress), task };
   298→   298→   298→}

   301→   301→   301→// دمج الفيديوهات — runMerge (القسم 7.3)

   303→   303→   303→export async function runMerge(args: Record<string, unknown>, notify?: NotifyFn): Promise<ToolOutcome> {
   304→   304→   304→  const urlsArg = Array.isArray(args.urls) ? (args.urls as string[]) : [];
   305→   305→   305→  const taskIdsArg = Array.isArray(args.task_ids) ? (args.task_ids as string[]) : [];
   306→   306→   306→
   307→   307→   307→  // 1) تحقق صارم: مجموع ≥2 و ≤10
   308→   308→   308→  if (urlsArg.length + taskIdsArg.length < 2) {
   309→   309→   309→    return { status: "params", isError: true, text: fmtMergeNeedsTwo() };
   310→   310→   310→  }
   311→   311→   311→  if (urlsArg.length + taskIdsArg.length > MERGER_LIMITS.maxClips) {

   526→   526→   526→
   527→   527→   527→  task = (await updateTask(task.id, { status: "failed", error_message: sanitize(msg) }))!;
   528→   528→   528→  return { status: "failed", isError: true, text: fmtFailed(sanitize(msg)), task };
   529→   529→   529→}

   540→   540→   540→async function refreshMergeTask(task: TaskRow): Promise<ToolOutcome> {

   558→   558→   558→    } catch (err) {

   564→   564→   564→
   565→   565→   565→  let info: MergerJobInfo;
   566→   566→   566→  try {
   567→   567→   567→    info = await getMergeJob(task.provider_task_id);
   568→   568→   568→  } catch (err) {
   569→   569→   569→    if (err instanceof ProviderError && (err as ProviderError & { mergerNotFound?: boolean }).mergerNotFound) {
   570→   570→   570→      // 404 → فشل نهائي: حُذفت أو انتهت صلاحيتها بعد 24 ساعة

   589→   589→   589→  }
   590→   590→   590→  task = (await updateTask(task.id, { status: "processing", progress: info.progress }))!;
