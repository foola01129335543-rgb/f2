    48→/** إخفاء أي نص طويل قد يحتوي مفاتيح من رسائل الأخطاء قبل عرضها للعميل */
    49→function sanitize(msg: string): string {
    50→  return String(msg ?? "سبب غير معروف").replace(/[A-Za-z0-9_\-./]{25,}/g, "***").slice(0, 300);
    51→}

    53→function effectiveWait(requested: number | undefined, fallback: number): number {
    54→  const w = typeof requested === "number" && requested >= 0 ? requested : fallback;
    55→  return Math.min(w, WAIT_CAP);
    56→}

    61→function buildImageInput(args: Record<string, unknown>): ImageInput {
    62→  const input: ImageInput = { prompt: String(args.prompt) };
    63→  const ar = args.aspect_ratio as string | undefined;
    64→  if (ar && ar !== "auto") input.aspect_ratio = ar;
    65→  if (args.resolution) input.resolution = String(args.resolution);
    66→  if (args.output_format) input.output_format = String(args.output_format);
    67→  // background يُطبق عند 1K فقط حسب المواصفة
    68→  if (args.background && args.background !== "auto" && (args.resolution ?? "1K") === "1K") {
    69→    input.background = String(args.background);
    70→  }
    71→  return input;
    72→}
    73→
    74→function buildVideoInput(args: Record<string, unknown>): VideoInput {
    75→  const input: VideoInput = { prompt: String(args.prompt) };
    76→  if (args.image_url) input.image_url = String(args.image_url);
    77→  const ar = args.aspect_ratio as string | undefined;
    78→  if (ar && ar !== "auto") input.aspect_ratio = ar;
    79→  if (args.resolution) input.resolution = String(args.resolution);
    80→  if (typeof args.enable_audio === "boolean") input.enable_audio = args.enable_audio;
    81→  if (typeof args.duration === "number") input.duration = Math.round(args.duration);
    82→  return input;
    83→}

    93→/**
    94→ * تجربة المرشحين بالترتيب حتى نجاح الإنشاء.
    95→ * كل محاولة (نجاحًا أو فشلًا) تُسجَّل في attempts_json وتحدّث عدّادات المفتاح.
    96→ * skipIds: مفاتيح يُتجاوزها هذا التشغيل فقط (المفتاح الذي فشل للتو في منتصف المهمة).
    97→ */
    98→async function tryCreateWithKeys(opts: {
    99→  task: TaskRow;
   100→  modelDef: ModelDef;
   101→  input: ImageInput | VideoInput;
   102→  skipIds?: Set<string>;
   103→}): Promise<CreateAttemptOutcome> {
   104→  const { task, modelDef, input, skipIds } = opts;
   105→  const pool = await getKeyPool(modelDef.providers);
   106→  const candidates = pool.filter((k) => !skipIds?.has(k.id));
   107→
   108→  for (const key of candidates) {
   109→    const adapter = getAdapter(key.provider);
   110→    const providerModel = modelDef.providerModel[key.provider] ?? modelDef.canonical;
   111→    try {
   112→      // صور متزامنة إن دعم المزود ذلك (CometAPI)
   113→      if (task.type === "image" && adapter.createImageSync) {
   114→        const { urls } = await adapter.createImageSync(key, input as ImageInput);
   115→        await bumpKeyCounters(key.id, true);
   116→        // مفتاح كان مستنزفًا وعاد للعمل (شُحن رصيده) → نعيده active
   117→        if (key.status === "exhausted") await setKeyStatus(key.id, "active", null);
   118→        await recordAttempt(task.id, { provider: key.provider, keyId: key.id, ok: true });
   119→        return { kind: "created_sync", key, syncUrls: urls };
   120→      }
   121→      const { providerTaskId } = await adapter.createTask(key, providerModel, input);
   122→      await bumpKeyCounters(key.id, true);
   123→      if (key.status === "exhausted") await setKeyStatus(key.id, "active", null);
   124→      await recordAttempt(task.id, { provider: key.provider, keyId: key.id, ok: true });
   125→      return { kind: "created", key, providerTaskId };
   126→    } catch (err) {
   127→      const cls = classifyProviderError(err);
   128→      const msg = err instanceof Error ? err.message : String(err);
   129→      await bumpKeyCounters(key.id, false, msg.slice(0, 240));
   130→      await recordAttempt(task.id, {
   131→        provider: key.provider, keyId: key.id, ok: false,
   132→        errorClass: cls, error: msg.slice(0, 240),
   133→      });
   134→      if (cls === "params") {
   135→        // خطأ معطيات — توقف فورًا: تبديل المفاتيح لن يحلّه
   136→        return { kind: "params", paramsMessage: sanitize(msg) };
   137→      }
   138→      if (cls === "disabled") {
   139→        await setKeyStatus(key.id, "disabled", msg.slice(0, 240)); // مفتاح مرفوض → تعطيل تلقائي
   140→        continue; // → المفتاح التالي
   141→      }
   142→      if (cls === "exhausted") {
   143→        await setKeyStatus(key.id, "exhausted", msg.slice(0, 240)); // يبقى في القاعدة
   144→        continue; // → المفتاح التالي
   145→      }
   146→      // provider (موديل غير مدعوم) أو temporary → التالي
   147→      continue;
   148→    }
   149→  }
   150→  return { kind: "exhausted" };
   151→}

   153→async function recordAttempt(taskId: string, a: Omit<AttemptRecord, "at">): Promise<void> {
   154→  // تُسجَّل المحاولة عبر updateTask بلا تعديل حقول أخرى
   155→  await updateTask(taskId, {}, { ...a, at: new Date().toISOString() });
   156→}

   161→export async function runGeneration(
   162→  type: "image" | "video",
   163→  args: Record<string, unknown>,
   164→  notify?: NotifyFn
   165→): Promise<ToolOutcome> {
   166→  const modelDef = resolveModel(args.model as string | undefined);
   167→  if (!modelDef || modelDef.type !== type) {
   168→    return {
   169→      status: "params",
   170→      isError: true,
   171→      text: fmtFailed(`موديل غير معروف: ${String(args.model ?? "")}. المتاح: gpt-image-2-6/flare-text-to-image (صور) و grok-imagine-video-1-5-preview (فيديو).`),
   172→    };
   173→  }
   174→
   175→  const pool = await getKeyPool(modelDef.providers);
   176→  if (!pool.length) return { status: "no_keys", isError: true, text: fmtNoKeys(modelDef.canonical) };
   177→
   178→  // 3) سجّل المهمة قبل أي نداء خارجي
   179→  const input = type === "image" ? buildImageInput(args) : buildVideoInput(args);
   180→  let task = await insertTask({
   181→    id: randomUUID(),
   182→    model: modelDef.canonical,
   183→    type,
   184→    prompt: input.prompt,
   185→    params_json: JSON.stringify(args),
   186→    status: "processing",
   187→  });
   188→
   189→  const waitSec = effectiveWait(args.wait_seconds as number | undefined, type === "image" ? 90 : 25);
   190→  let deadline = Date.now() + waitSec * 1000;
   191→
   192→  // 4) تجربة المرشحين بالترتيب حتى نجاح الإنشاء
   193→  let created = await tryCreateWithKeys({ task, modelDef, input });
   194→  if (created.kind === "params") {
   195→    task = (await updateTask(task.id, { status: "failed", error_message: created.paramsMessage }))!;
   196→    return { status: "failed", isError: true, text: fmtFailed(created.paramsMessage!), task };
   197→  }
   198→  if (created.kind === "exhausted") {
   199→    // تبقى processing — تُعاد المحاولة مع كل get_task بعد التهدئة (لا شيء يُذكر للعميل)
   200→    task = (await updateTask(task.id, { error_message: "بانتظار توفر مفتاح صالح…" }))!;
   201→    return { status: "processing", text: fmtProcessing(task.id), task };
   202→  }
   203→
   204→  if (created.kind === "created_sync") {
   205→    task = (await updateTask(task.id, {
   206→      provider: created.key!.provider, key_id: created.key!.id,
   207→      status: "completed", result_urls_json: JSON.stringify(created.syncUrls),
   208→    }))!;
   209→    notify?.(100, 100, "اكتمل التوليد").catch(() => {});
   210→    return { status: "completed", text: fmtImageDone(created.syncUrls!), task };
   211→  }
   212→
   213→  // نجح الإنشاء — اربط المهمة بالمزود والمفتاح
   214→  task = (await updateTask(task.id, {
   215→    provider: created.key!.provider,
   216→    provider_task_id: created.providerTaskId,
   217→    key_id: created.key!.id,
   218→    progress: 5,
   219→  }))!;
   220→  let providerTaskId = created.providerTaskId!;
   221→  let adapter = getAdapter(created.key!.provider);
   222→  let currentKey = created.key!;
   223→
   224→  // 5) نافذة الانتظار: استعلام كل 3 ثوانٍ مع بث تقدم حي
   225→  while (Date.now() < deadline) {
   226→    let check;
   227→    try {
   228→      check = await adapter.checkTask(currentKey, providerTaskId);
   229→    } catch (err) {
   230→      // أخطاء شبكتنا نحن (timeout/fetch) = مؤقتة → استمر بالاستعلام
   231→      if (!(err instanceof ProviderError) || err.status !== 0) {
   232→        // خطأ مصنَّف من المزود أثناء الاستعلام: عالجه كفشل منتصف مهمة
   233→        check = { status: "failed", error: err instanceof Error ? err.message : String(err) };
   234→      } else {
   235→        await sleep(POLL_INTERVAL_MS);
   236→        continue;
   237→      }
   238→    }
   239→
   240→    if (check.status === "processing") {
   241→      const p = check.progress ?? task.progress;
   242→      if (Number.isFinite(p) && p > task.progress) task = (await updateTask(task.id, { progress: Math.round(p) })) ?? task;
   243→      notify?.(Math.round(check.progress ?? task.progress), 100, `قيد التوليد… ${Math.round(check.progress ?? task.progress)}%`).catch(() => {});
   244→      await sleep(POLL_INTERVAL_MS);
   245→      continue;
   246→    }
   247→
   248→    if (check.status === "completed" && check.urls?.length) {
   249→      task = (await updateTask(task.id, { status: "completed", result_urls_json: JSON.stringify(check.urls) }))!;
   250→      notify?.(100, 100, "اكتمل التوليد").catch(() => {});
   251→      return {
   252→        status: "completed",
   253→        text: type === "image" ? fmtImageDone(check.urls) : fmtVideoDone(check.urls),
   254→        task,
   255→      };
   256→    }
   257→
   258→    if (check.status === "failed") {
   259→      const msg = String(check.error ?? "");
   260→      if (looksLikeKeyIssue(msg)) {
   261→        // فشل مفتاح/رصيد في منتصف المهمة → أعد إنشاء نفس المهمة بنفس task_id على المفتاح التالي
   262→        const skip = new Set<string>([currentKey.id]);
   263→        const retry = await tryCreateWithKeys({ task, modelDef, input, skipIds: skip });
   264→        if (retry.kind === "created" || retry.kind === "created_sync") {
   265→          task = (await updateTask(task.id, {
   266→            provider: retry.key!.provider,
   267→            key_id: retry.key!.id,
   268→            provider_task_id: retry.providerTaskId ?? task.provider_task_id,
   269→            error_message: null,
   270→            progress: 5,
   271→          }))!;
   272→          if (retry.kind === "created_sync") {
   273→            task = (await updateTask(task.id, { status: "completed", result_urls_json: JSON.stringify(retry.syncUrls) }))!;
   274→            return { status: "completed", text: fmtImageDone(retry.syncUrls!), task };
   275→          }
   276→          providerTaskId = retry.providerTaskId!;
   277→          adapter = getAdapter(retry.key!.provider);
   278→          currentKey = retry.key!;
   279→          deadline = Date.now() + waitSec * 1000; // نافذة جديدة للمحاولة الجديدة
   280→          continue;
   281→        }
   282→        if (retry.kind === "params") {
   283→          task = (await updateTask(task.id, { status: "failed", error_message: retry.paramsMessage }))!;
   284→          return { status: "failed", isError: true, text: fmtFailed(retry.paramsMessage!), task };
   285→        }
   286→        // لا مفتاح متاح الآن → تبقى processing بصمت
   287→        task = (await updateTask(task.id, { error_message: "بانتظار توفر مفتاح صالح…" }))!;
   288→        return { status: "processing", text: fmtProcessing(task.id), task };
   289→      }
   290→      // فشل حقيقي (انتهاك سياسة/معطيات) → failed
   291→      task = (await updateTask(task.id, { status: "failed", error_message: sanitize(msg) }))!;
   292→      return { status: "failed", isError: true, text: fmtFailed(sanitize(msg)), task };
   293→    }
   294→  }
   295→
   296→  // 7) انتهت النافذة والنتيجة لم تظهر → أعد المهمة بحالة processing ليتابعها العميل
   297→  return { status: "processing", text: fmtProcessing(task.id, task.progress), task };
   298→}

   303→export async function runMerge(args: Record<string, unknown>, notify?: NotifyFn): Promise<ToolOutcome> {
   304→  const urlsArg = Array.isArray(args.urls) ? (args.urls as string[]) : [];
   305→  const taskIdsArg = Array.isArray(args.task_ids) ? (args.task_ids as string[]) : [];
   306→
   307→  // 1) تحقق صارم: مجموع ≥2 و ≤10
   308→  if (urlsArg.length + taskIdsArg.length < 2) {
   309→    return { status: "params", isError: true, text: fmtMergeNeedsTwo() };
   310→  }
   311→  if (urlsArg.length + taskIdsArg.length > MERGER_LIMITS.maxClips) {
   312→    return { status: "params", isError: true, text: fmtMergeMax(MERGER_LIMITS.maxClips) };
   313→  }
   314→
   315→  // 2) حل روابط task_ids أولًا (مهام فيديو مكتملة بالترتيب) ثم urls المباشرة
   316→  const videos: string[] = [];
   317→  const failedDetails: string[] = [];
   318→  for (const tid of taskIdsArg) {
   319→    const t = await getTask(String(tid));
   320→    if (!t || t.type !== "video") failedDetails.push(`• ${tid}: مهمة غير موجودة أو ليست مهمة فيديو`);
   321→    else if (t.status !== "completed") failedDetails.push(`• ${tid}: المهمة ليست مكتملة بعد (حالتها ${t.status})`);
   322→    else {
   323→      const urls = safeParse<string[]>(t.result_urls_json, []);
   324→      if (!urls.length) failedDetails.push(`• ${tid}: مهمة مكتملة بلا روابط`);
   325→      else videos.push(urls[0]);
   326→    }
   327→  }
   328→  videos.push(...urlsArg.map(String));
   329→
   330→  if (videos.length < 2) {
   331→    return { status: "params", isError: true, text: fmtMergeNeedsTwo(failedDetails.join("\n")) };
   332→  }
   333→  // كل الروابط http/https مباشرة وعامة
   334→  const bad = videos.find((u) => !/^https?:\/\//i.test(u));
   335→  if (bad) return { status: "params", isError: true, text: fmtFailed(`الرابط غير صالح للدمج (يجب أن يكون http/https مباشرًا وعامًا): ${bad.slice(0, 120)}`) };
   336→
   337→  // 3) سجّل المهمة أولًا
   338→  let task = await insertTask({
   339→    id: randomUUID(),
   340→    provider: "merger",
   341→    model: "video-merger",
   342→    type: "merge",
   343→    prompt: `دمج ${videos.length} مقاطع`,
   344→    params_json: JSON.stringify(args),
   345→    status: "processing",
   346→  });
   347→
   348→  // 4) أنشئ مهمة الدمج لدى الخدمة — أي خطأ معلن = فشل حتمي فوري
   349→  let jobId: string;
   350→  try {
   351→    const createdJob = await createMergeJob(videos);
   352→    jobId = createdJob.jobId;
   353→  } catch (err) {
   354→    const msg = sanitize(err instanceof Error ? err.message : String(err));
   355→    task = (await updateTask(task.id, { status: "failed", error_message: msg }))!;
   356→    return { status: "failed", isError: true, text: fmtFailed(msg), task };
   357→  }
   358→  task = (await updateTask(task.id, { provider_task_id: jobId }))!;
   359→
   360→  // 5) أطلق process — يُنفَّذ مرة واحدة فقط (القفل داخل عميل الدمج)
   361→  const processPromise = processMergeJob(jobId).catch(() => null);
   362→
   363→  const waitSec = effectiveWait(args.wait_seconds as number | undefined, 60);
   364→  const deadline = Date.now() + waitSec * 1000;
   365→
   366→  // 6) استعلم كل 3 ثوانٍ مع بث تقدم — تحديث الحالة يُكتب في Turso فورًا
   367→  while (Date.now() < deadline) {
   368→    let info: MergerJobInfo;
   369→    try {
   370→      info = await getMergeJob(jobId);
   371→    } catch (err) {
   372→      if (err instanceof ProviderError && (err as ProviderError & { mergerNotFound?: boolean }).mergerNotFound) {
   373→        task = (await updateTask(task.id, { status: "failed", error_message: err.message }))!;
   374→        return { status: "failed", isError: true, text: fmtFailed(err.message), task };
   375→      }
   376→      await sleep(POLL_INTERVAL_MS); // شبكتنا نحن → مؤقت
   377→      continue;
   378→    }
   379→
   380→    if (info.status === "done" && (info.videoUrl || info.downloadUrl)) {
   381→      const urls = [info.videoUrl ?? info.downloadUrl!];
   382→      task = (await updateTask(task.id, { status: "completed", result_urls_json: JSON.stringify(urls) }))!;
   383→      notify?.(100, 100, "اكتمل الدمج").catch(() => {});
   384→      void processPromise; // اكتملنا قبل انتهاء انتظار process — لا شيء مطلوب
   385→      return { status: "completed", text: fmtMergeDone(urls), task };
   386→    }
   387→    if (info.status === "failed") {
   388→      // خطأ معلن من الخدمة = فشل نهائي فوري — لا إعادة محاولة (قاعدة حتمية)
   389→      task = (await updateTask(task.id, { status: "failed", error_message: sanitize(info.error ?? "فشل الدمج") }))!;
   390→      void processPromise;
   391→      return { status: "failed", isError: true, text: fmtFailed(sanitize(info.error ?? "فشل الدمج")), task };
   392→    }
   393→
   394→    task = (await updateTask(task.id, { status: "processing", progress: info.progress }))!;
   395→    notify?.(info.progress, 100, `جارٍ الدمج… ${info.progress}%`).catch(() => {});
   396→    await sleep(POLL_INTERVAL_MS);
   397→  }
   398→
   399→  void processPromise;
   400→  return { status: "processing", text: fmtProcessing(task.id, task.progress), task };
   401→}

   404→// تتبع مهمة — refreshTask (get_task) مع نفس منطق التبديل أعلاه
   405→// -----------------------------------------------------------------------------
   406→export async function refreshTask(taskId: string): Promise<ToolOutcome> {
   407→  let task = await getTask(taskId);
   408→  if (!task) return { status: "failed", isError: true, text: fmtTaskNotFound(taskId) };
   409→
   410→  if (task.status === "completed") {
   411→    const urls = safeParse<string[]>(task.result_urls_json, []);
   412→    return { status: "completed", text: fmtGetTask({ ...task, urls }), task };
   413→  }
   414→  if (task.status === "failed") {
   415→    return { status: "failed", isError: true, text: fmtGetTask({ ...task, urls: [] }), task };
   416→  }
   417→
   418→  // جارية — حدّثها حية
   419→  if (task.type === "merge") return refreshMergeTask(task);
   420→
   421→  const modelDef = resolveModel(task.model);
   422→  if (!modelDef) {
   423→    task = (await updateTask(task.id, { status: "failed", error_message: "موديل المهمة غير معروف" }))!;
   424→    return { status: "failed", isError: true, text: fmtFailed(task.error_message!), task };
   425→  }
   426→
   427→  if (!task.provider || !task.provider_task_id) {
   428→    // لم يُنشأ بعد على أي مزود (فشل على كل المفاتيح) — أعد المحاولة بعد تهدئة 45 ثانية
   429→    const since = Date.now() - new Date(task.updated_at.replace(" ", "T") + "Z").getTime();
   430→    if (since < CREATE_RETRY_COOLDOWN_MS) {
   431→      return { status: "processing", text: fmtProcessing(task.id, task.progress), task };
   432→    }
   433→    const input = task.type === "image" ? buildImageInput(safeParse(task.params_json, {})) : buildVideoInput(safeParse(task.params_json, {}));
   434→    const created = await tryCreateWithKeys({ task, modelDef, input });
   435→    if (created.kind === "created" || created.kind === "created_sync") {
   436→      task = (await updateTask(task.id, {
   437→        provider: created.key!.provider, key_id: created.key!.id,
   438→        provider_task_id: created.providerTaskId ?? null,
   439→        error_message: null, progress: 5,
   440→      }))!;
   441→      if (created.kind === "created_sync") {
   442→        task = (await updateTask(task.id, { status: "completed", result_urls_json: JSON.stringify(created.syncUrls) }))!;
   443→        return { status: "completed", text: fmtImageDone(created.syncUrls!), task };
   444→      }
   445→      return { status: "processing", text: fmtProcessing(task.id, 5), task };
   446→    }
   447→    if (created.kind === "params") {
   448→      task = (await updateTask(task.id, { status: "failed", error_message: created.paramsMessage }))!;
   449→      return { status: "failed", isError: true, text: fmtFailed(created.paramsMessage!), task };
   450→    }
   451→    // ما زلنا بلا مفاتيح صالحة — اضبط التهدئة من جديد (لا شيء يُذكر للعميل)
   452→    task = (await updateTask(task.id, { error_message: "بانتظار توفر مفتاح صالح…" }))!;
   453→    return { status: "processing", text: fmtProcessing(task.id), task };
   454→  }
   455→
   456→  // مهمة جارية لدى مزود — استعلم حالتها
   457→  const adapter = getAdapter(task.provider as ProviderId);
   458→  let keyRow = await findKeyById(task.key_id);
   459→  if (!keyRow && task.provider_task_id) {
   460→    // المفتاح حُذف أو غير مسجّل لكن المهمة قائمة لدى المزود — استعلام فقط بأي مفتاح متاح
   461→    const pool = await getKeyPool([task.provider as ProviderId]);
   462→    if (pool.length) keyRow = pool[0];
   463→  }
   464→  if (!keyRow) {
   465→    // لا مفتاح للاستعلام ولا مهمة قائمة — أعد الإنشاء على أي مفتاح متاح
   466→    const input = task.type === "image" ? buildImageInput(safeParse(task.params_json, {})) : buildVideoInput(safeParse(task.params_json, {}));
   467→    const created = await tryCreateWithKeys({ task, modelDef, input });
   468→    if (created.kind === "created" || created.kind === "created_sync") {
   469→      task = (await updateTask(task.id, {
   470→        provider: created.key!.provider, key_id: created.key!.id,
   471→        provider_task_id: created.providerTaskId ?? null, progress: 5, error_message: null,
   472→      }))!;
   473→      if (created.kind === "created_sync") {
   474→        task = (await updateTask(task.id, { status: "completed", result_urls_json: JSON.stringify(created.syncUrls) }))!;
   475→        return { status: "completed", text: fmtImageDone(created.syncUrls!), task };
   476→      }
   477→      return { status: "processing", text: fmtProcessing(task.id, 5), task };
   478→    }
   479→    return { status: "processing", text: fmtProcessing(task.id, task.progress), task };
   480→  }
   481→
   482→  let check;
   483→  try {
   484→    check = await adapter.checkTask(keyRow, task.provider_task_id!);
   485→  } catch (err) {
   486→    if (err instanceof ProviderError && err.status === 0) {
   487→      return { status: "processing", text: fmtProcessing(task.id, task.progress), task }; // شبكة → مؤقت
   488→    }
   489→    check = { status: "failed", error: err instanceof Error ? err.message : String(err) };
   490→  }
   491→
   492→  if (check.status === "processing") {
   493→    const p = check.progress ?? task.progress;
   494→    if (Number.isFinite(p) && p > task.progress) task = (await updateTask(task.id, { progress: Math.round(p) })) ?? task;
   495→    return { status: "processing", text: fmtProcessing(task.id, task.progress), task };
   496→  }
   497→
   498→  if (check.status === "completed" && check.urls?.length) {
   499→    task = (await updateTask(task.id, { status: "completed", result_urls_json: JSON.stringify(check.urls) }))!;
   500→    return { status: "completed", text: fmtGetTask({ ...task, urls: check.urls }), task };
   501→  }
   502→
   503→  // فشل منتصف مهمة — نفس منطق التبديل
   504→  const msg = String(check.error ?? "");
   505→  if (looksLikeKeyIssue(msg)) {
   506→    const input = task.type === "image" ? buildImageInput(safeParse(task.params_json, {})) : buildVideoInput(safeParse(task.params_json, {}));
   507→    const created = await tryCreateWithKeys({ task, modelDef, input, skipIds: new Set([keyRow.id]) });
   508→    if (created.kind === "created" || created.kind === "created_sync") {
   509→      task = (await updateTask(task.id, {
   510→        provider: created.key!.provider, key_id: created.key!.id,
   511→        provider_task_id: created.providerTaskId ?? null, progress: 5, error_message: null,
   512→      }))!;
   513→      if (created.kind === "created_sync") {
   514→        task = (await updateTask(task.id, { status: "completed", result_urls_json: JSON.stringify(created.syncUrls) }))!;
   515→        return { status: "completed", text: fmtImageDone(created.syncUrls!), task };
   516→      }
   517→      return { status: "processing", text: fmtProcessing(task.id, 5), task };
   518→    }
   519→    if (created.kind === "params") {
   520→      task = (await updateTask(task.id, { status: "failed", error_message: created.paramsMessage }))!;
   521→      return { status: "failed", isError: true, text: fmtFailed(created.paramsMessage!), task };
   522→    }
   523→    task = (await updateTask(task.id, { error_message: "بانتظار توفر مفتاح صالح…" }))!;
   524→    return { status: "processing", text: fmtProcessing(task.id), task };
   525→  }
   526→
   527→  task = (await updateTask(task.id, { status: "failed", error_message: sanitize(msg) }))!;
   528→  return { status: "failed", isError: true, text: fmtFailed(sanitize(msg)), task };
   529→}

   538→// تتبع مهمة دمج — refreshMergeTask (القسم 7.3.6)
   539→// -----------------------------------------------------------------------------
   540→async function refreshMergeTask(task: TaskRow): Promise<ToolOutcome> {
   541→  if (!task.provider_task_id) {
   542→    // فشل إنشاء مهمة الدمج سابقًا — أعد المحاولة بعد التهدئة
   543→    const since = Date.now() - new Date(task.updated_at.replace(" ", "T") + "Z").getTime();
   544→    if (since < CREATE_RETRY_COOLDOWN_MS) {
   545→      return { status: "processing", text: fmtProcessing(task.id, task.progress), task };
   546→    }
   547→    const params = safeParse<Record<string, unknown>>(task.params_json, {});
   548→    const vids = Array.isArray(params.urls) ? (params.urls as string[]).map(String) : [];
   549→    if (vids.length < 2) {
   550→      task = (await updateTask(task.id, { status: "failed", error_message: "روابط الدمج الأصلية غير متوفرة" }))!;
   551→      return { status: "failed", isError: true, text: fmtFailed(task.error_message!), task };
   552→    }
   553→    try {
   554→      const createdJob = await createMergeJob(vids);
   555→      task = (await updateTask(task.id, { provider_task_id: createdJob.jobId, error_message: null }))!;
   556→      void processMergeJob(createdJob.jobId).catch(() => null);
   557→      return { status: "processing", text: fmtProcessing(task.id, task.progress), task };
   558→    } catch (err) {
   559→      const msg = sanitize(err instanceof Error ? err.message : String(err));
   560→      task = (await updateTask(task.id, { status: "failed", error_message: msg }))!;
   561→      return { status: "failed", isError: true, text: fmtFailed(msg), task };
   562→    }
   563→  }
   564→
   565→  let info: MergerJobInfo;
   566→  try {
   567→    info = await getMergeJob(task.provider_task_id);
   568→  } catch (err) {
   569→    if (err instanceof ProviderError && (err as ProviderError & { mergerNotFound?: boolean }).mergerNotFound) {
   570→      // 404 → فشل نهائي: حُذفت أو انتهت صلاحيتها بعد 24 ساعة
   571→      task = (await updateTask(task.id, { status: "failed", error_message: err.message }))!;
   572→      return { status: "failed", isError: true, text: fmtFailed(err.message), task };
   573→    }
   574→    return { status: "processing", text: fmtProcessing(task.id, task.progress), task }; // شبكتنا → مؤقت
   575→  }
   576→
   577→  if (info.status === "done" && (info.videoUrl || info.downloadUrl)) {
   578→    const urls = [info.videoUrl ?? info.downloadUrl!];
   579→    task = (await updateTask(task.id, { status: "completed", result_urls_json: JSON.stringify(urls) }))!;
   580→    return { status: "completed", text: fmtMergeDone(urls), task };
   581→  }
   582→  if (info.status === "failed") {
   583→    task = (await updateTask(task.id, { status: "failed", error_message: sanitize(info.error ?? "فشل الدمج") }))!;
   584→    return { status: "failed", isError: true, text: fmtFailed(sanitize(info.error ?? "فشل الدمج")), task };
   585→  }
   586→  if (info.status === "queued") {
   587→    // أُنشئت المهمة ولم يُنفَّذ process بعد → نفّذه الآن (القفل يمنع التزامن)
   588→    void processMergeJob(task.provider_task_id).catch(() => null);
   589→  }
   590→  task = (await updateTask(task.id, { status: "processing", progress: info.progress }))!;
   591→  return { status: "processing", text: fmtProcessing(task.id, info.progress), task };
   592→}
